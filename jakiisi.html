(() => { const body = document.body; const sidebar = document.getElementById("site-sidebar"); const menu = document.querySelector(".mobile-menu"); const overlay = document.querySelector(".sidebar-overlay"); function openNav() { body.classList.add("nav-open"); menu?.setAttribute("aria-expanded", "true"); } function closeNav() { body.classList.remove("nav-open"); menu?.setAttribute("aria-expanded", "false"); } menu?.addEventListener("click", openNav); overlay?.addEventListener("click", closeNav); sidebar?.querySelector(".sidebar-close")?.addEventListener("click", closeNav); sidebar?.querySelectorAll("a").forEach(a => a.addEventListener("click", closeNav)); document.addEventListener("keydown", e => { if (e.key === "Escape") closeNav(); }); // Highlight same-page hash links when one is active. function markHashLink() { const file = location.pathname.split("/").pop() || "index.html"; sidebar?.querySelectorAll(".sidebar-links a").forEach(a => { const url = new URL(a.href, location.href); const sameFile = (url.pathname.split("/").pop() || "index.html") === file; const isHashMatch = sameFile && url.hash && url.hash === location.hash; if (isHashMatch) { sidebar.querySelectorAll(".sidebar-links a.active").forEach(x => { x.classList.remove("active"); x.removeAttribute("aria-current"); }); a.classList.add("active"); a.setAttribute("aria-current", "page"); } }); } window.addEventListener("hashchange", markHashLink); markHashLink(); // Tooltip is appended to BODY, not the datasheet. This prevents clipping/layout weirdness. const popover = document.getElementById("keyword-tooltip-popover"); let active = null; function positionPopover(el) { if (!popover || popover.hidden) return; const r = el.getBoundingClientRect(); const pr = popover.getBoundingClientRect(); const pad = 10; let left = r.left + r.width / 2 - pr.width / 2; left = Math.max(pad, Math.min(left, window.innerWidth - pr.width - pad)); let top = r.top - pr.height - 9; if (top < pad) top = r.bottom + 9; top = Math.max(pad, Math.min(top, window.innerHeight - pr.height - pad)); popover.style.left = `${Math.round(left)}px`; popover.style.top = `${Math.round(top)}px`; } function showTooltip(el) { if (!popover) return; const text = el.dataset.tooltip; if (!text) return; active = el; // Remove the native title only while the custom tooltip system is alive. if (el.hasAttribute("title")) { el.dataset.nativeTitle = el.getAttribute("title"); el.removeAttribute("title"); } popover.textContent = text; popover.hidden = false; positionPopover(el); } function hideTooltip(el) { if (!popover || (el && active !== el)) return; popover.hidden = true; popover.textContent = ""; active = null; } document.querySelectorAll(".keyword-tip[data-tooltip]").forEach(el => { el.addEventListener("mouseenter", () => showTooltip(el)); el.addEventListener("mouseleave", () => hideTooltip(el)); el.addEventListener("focus", () => showTooltip(el)); el.addEventListener("blur", () => hideTooltip(el)); el.addEventListener("click", e => { // Useful on touch screens. if (active === el && !popover.hidden) { hideTooltip(el); } else { e.preventDefault(); showTooltip(el); } }); }); window.addEventListener("resize", () => active && positionPopover(active)); window.addEventListener("scroll", () => active && positionPopover(active), true); })();
